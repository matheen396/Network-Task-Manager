cmake_minimum_required(VERSION 3.20)
project(netask VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(netask_core src/task.cpp src/task_store.cpp src/task_manager.cpp src/protocol.cpp)
target_include_directories(netask_core PUBLIC include)
target_link_libraries(netask_core PUBLIC Threads::Threads)
if(WIN32)
  target_link_libraries(netask_core PUBLIC ws2_32)
endif()
if(MSVC)
  target_compile_options(netask_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(netask_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()

add_executable(netask_server src/server_main.cpp src/tcp_server.cpp)
target_link_libraries(netask_server PRIVATE netask_core)
add_executable(netask src/client_main.cpp src/tcp_client.cpp)
target_link_libraries(netask PRIVATE netask_core)

include(CTest)
if(BUILD_TESTING)
  add_executable(netask_tests tests/test_main.cpp)
  target_link_libraries(netask_tests PRIVATE netask_core)
  add_test(NAME netask_tests COMMAND netask_tests)
endif()
